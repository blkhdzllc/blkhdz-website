async function switchTab(fileName, el) { document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active')); el.classList.add('active'); const container = document.getElementById('market-grid'); container.innerHTML = '
SCANNING DATABASE...
'; try { const response = await fetch('data/inventory.json?v=' + Date.now()); const fullData = await response.json(); // Isolate data based on category mapping const isLego = (fileName === 'inventory.json'); const items = isLego ? fullData.lego : fullData.diecast; if (!items || items.length === 0) { container.innerHTML = '
NO ACTIVE ENTRIES IN CATEGORY.
'; return; } container.innerHTML = items.map(item => { const price = item.price ? `$${item.price}` : 'VIEW ON EBAY'; const skuColor = isLego ? 'var(--blk-red)' : 'var(--blk-teal)'; const skuLabel = isLego ? 'SET' : 'SKU'; return `
${skuLabel} #${item.id}
${item.name}
${price}
 `; }).join(''); } catch (error) { container.innerHTML = '
CATALOG ENGINE FAULT
'; } }
